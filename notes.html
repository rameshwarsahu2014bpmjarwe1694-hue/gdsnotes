require('dotenv').config(); const multer = require('multer'); const upload = multer({ dest: 'uploads/' }); const path = require('path'); const express = require('express'); const fs = require('fs'); const admin = require('firebase-admin'); const cors = require('cors'); const axios = require('axios'); const crypto = require('crypto'); const app = express(); // --- 1. MIDDLEWARES --- app.use(cors()); app.use(express.json()); app.use((req, res, next) => { res.setHeader("X-Content-Type-Options", "nosniff"); res.setHeader("X-Frame-Options", "DENY"); res.setHeader("X-XSS-Protection", "1; mode=block"); next(); }); // --- 2. FIREBASE SETUP --- const keyPath = path.join(process.cwd(), 'serviceAccountKey.json'); if (fs.existsSync(keyPath)) { if (!admin.apps.length) { admin.initializeApp({ credential: admin.credential.cert(require(keyPath)) }); } console.log("Firebase Connected Successfully!"); } else { console.error("CRITICAL ERROR: serviceAccountKey.json nahi mili!"); } const db = admin.firestore(); async function verifyFirebaseUser(req, res, next) { try { const authHeader = req.headers.authorization || ""; if (!authHeader.startsWith("Bearer ")) { return res.status(401).json({ error: "TOKEN_REQUIRED" }); } const idToken = authHeader.split("Bearer ")[1]; const decoded = await admin.auth().verifyIdToken(idToken); req.user = decoded; console.log(`[AUTH] User: ${decoded.email}`); next(); } catch (err) { console.error("Auth Error:", err); return res.status(401).json({ error: "INVALID_TOKEN" }); } } // --- 3. HELPER FUNCTION --- const isTransactionProcessed = async (email, txnId) => { const doc = await db.collection('payments') .doc(email) .collection('transactions') .doc(txnId) .get(); return doc.exists && doc.data().status === "completed"; }; const findFileDynamically = (folderPath, year, post, subject) => { try { if (!fs.existsSync(folderPath)) return null; const files = fs.readdirSync(folderPath); // Agar 'year' Notes hai ya missing hai, toh Subject ke naam se file dhundo if (!year || year === "Notes" || year === "") { return files.find(file => file.toLowerCase().includes((subject || "").toLowerCase().replace(/ /g, '-'))); } // PYQ ke liye purana logic const cleanYear = year.replace('PYQ ', ''); const regex = new RegExp(`(?=.*${cleanYear})(?=.*${post})`, 'i'); return files.find(file => regex.test(file)); } catch (error) { console.error("Search Error:", error); return null; } }; const isValidInput = (value) => { if (!value) return false; if (typeof value !== "string") return false; // block dangerous patterns if (value.includes("..") || value.includes("/") || value.includes("\\")) { return false; } return true; }; const clean = (str) => str.toLowerCase().replace(/[^a-z0-9]/g, ""); // --- 4. ROUTE: GET PDF (PYQ Free / Notes Paid) --- app.get('/get-pdf', verifyFirebaseUser, async (req, res) => { const referer = req.headers.referer || ''; const secretHeader = req.headers['x-source-auth']; if (!referer.includes('github.io') && secretHeader !== 'RSH-GDS-SECURE-ACCESS') { return res.status(403).send("Unauthorized Access"); } const { year, post, subject } = req.query; const email = req.user.email; if (!isValidInput(post) || !isValidInput(subject || "")) { return res.status(400).json({ error: "INVALID_INPUT" }); } if (!email || !post) { return res.status(400).json({ error: "Email aur Subject zaroori hain." }); } try { if (!year || year === "undefined" || year === "" || year === "Notes") { const paymentDoc = await db.collection('payments').doc(email).get(); console.log(`[PAYMENT-CHECK] User: ${email}, Status: ${paymentDoc.data()?.status}`); if (!paymentDoc.exists || paymentDoc.data().status !== "completed") { return res.status(403).json({ error: "PAYMENT_REQUIRED" }); } } const pdfsDirectory = path.join(process.cwd(), 'pdfs'); const matchedFileName = findFileDynamically(pdfsDirectory, year || "", post, subject || ""); if (matchedFileName) { const filePath = path.join(pdfsDirectory, matchedFileName); res.setHeader('Content-Type', 'application/pdf'); const safeFile = path.basename(filePath); const finalPath = path.join(pdfsDirectory, safeFile); console.log(`[ACCESS] File found: ${matchedFileName}`); return res.sendFile(finalPath, (err) => { if (err) { console.error("File Send Error:", err); return res.status(500).json({ error: "FILE_ERROR" }); } }); } else { console.log(`[ACCESS] File NOT FOUND for: ${post}, ${subject}`); return res.status(404).json({ error: "File nahi mili server par." }); } } catch (err) { console.error("GET Error:", err); res.status(500).json({ error: "Internal Server Error" }); } }); // --- 5. PAYMENT ROUTES (Sirf Notes ke liye kaam aayenge) --- const PHONEPE_MID = process.env.PHONEPE_MID || "YOUR_MID"; const PHONEPE_SALT_KEY = process.env.PHONEPE_SALT_KEY || "YOUR_SALT_KEY"; const PHONEPE_SALT_INDEX = "1"; const PHONEPE_API_URL = "https://api.phonepe.com/apis/hermes/pg/v1/pay"; app.post('/submit-proof', verifyFirebaseUser, upload.single('screenshot'), async (req, res) => { try { const { txnId, email } = req.body; console.log(`[PROOF] Submission by: ${email}, Txn: ${txnId}`); const file = req.file; // Ye uploaded file hai // Firestore mein save karein await db.collection('payments') .doc(email) .collection('transactions') .doc(txnId || "MANUAL_" + Date.now()) .set({ status: "pending", transactionId: txnId, screenshotPath: file ? file.path : null, submittedAt: admin.firestore.FieldValue.serverTimestamp() }); return res.json({ success: true }); } catch (err) { console.error("Submit Proof Error:", err); res.status(500).json({ error: "Internal Server Error" }); } }); app.post('/phonepe-webhook', async (req, res) => { try { const rawBody = JSON.stringify(req.body); const xVerify = req.headers['x-verify']; const generatedSignature = crypto .createHash('sha256') .update(rawBody + PHONEPE_SALT_KEY) .digest('hex'); if (xVerify !== generatedSignature) { return res.status(401).send("Invalid Signature"); } const { merchantTransactionId, state } = req.body; if (!merchantTransactionId) { return res.status(400).send("Invalid webhook"); } // SAFE global transaction mapping (no email dependency) const txnRef = db.collection('payments') .doc("GLOBAL") .collection('transactions') .doc(merchantTransactionId); const doc = await txnRef.get(); if (doc.exists && doc.data().status === "completed") { return res.status(200).send("Already Processed"); } if (state === "COMPLETED") { await txnRef.set({ status: "completed", verifiedVia: "webhook", paidAt: admin.firestore.FieldValue.serverTimestamp() }, { merge: true }); } return res.status(200).send("OK"); } catch (err) { console.error("Webhook Error:", err); return res.status(500).send("ERROR"); } }); async function verifyPhonePePayment(txnId) { const merchantId = PHONEPE_MID; const apiPath = `/pg/v1/status/${merchantId}/${txnId}`; const checksum = crypto .createHash('sha256') .update(apiPath + PHONEPE_SALT_KEY) .digest('hex') + "###" + PHONEPE_SALT_INDEX; const response = await axios.get( `https://api.phonepe.com/apis/hermes${apiPath}`, { headers: { "Content-Type": "application/json", "X-VERIFY": checksum, "X-MERCHANT-ID": merchantId } } ); return response.data; } app.post('/initiate-payment', verifyFirebaseUser, async (req, res) => { try { const { email, amount, year, post, subject } = req.body; if (req.user.email !== email) { return res.status(403).json({ error: "EMAIL_MISMATCH" }); } const transactionId = "TXN" + Date.now(); await db.collection('payments') .doc(email) .collection('transactions') .doc(transactionId) .set({ status: "pending", amount, requested_for: `${year} - ${post} - ${subject}`, submittedAt: admin.firestore.FieldValue.serverTimestamp() }); const payload = { merchantId: PHONEPE_MID, merchantTransactionId: transactionId, merchantUserId: email.replace(/[^a-zA-Z0-9]/g, ''), amount: amount * 100, redirectUrl: `https://payment-backend-2-g6go.onrender.com/payment-status?txnId=${transactionId}&email=${email}&year=${year}&post=${post}&subject=${subject}`, redirectMode: "POST", paymentInstrument: { type: "PAY_PAGE" } }; const base64Payload = Buffer.from(JSON.stringify(payload)).toString('base64'); const fullURL = base64Payload + "/pg/v1/pay" + PHONEPE_SALT_KEY; const checksum = crypto.createHash('sha256').update(fullURL).digest('hex') + "###" + PHONEPE_SALT_INDEX; const response = await axios.post(PHONEPE_API_URL, { request: base64Payload }, { headers: { 'Content-Type': 'application/json', 'X-VERIFY': checksum, 'accept': 'application/json' } } ); res.json({ url: response.data.data.instrumentResponse.redirectInfo.url }); } catch (error) { console.error("Payment Error:", error.message); res.status(500).json({ error: "Payment initiation failed" }); } }); // --- FIX: PAYMENT STATUS ROUTES (POST for Frontend, GET for PhonePe Redirect) --- // 1.
